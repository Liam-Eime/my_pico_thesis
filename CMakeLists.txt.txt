cmake_minimum_required(VERSION 3.13)

# import the Pico SDK; this uses PICO_SDK_PATH environment or relative path
include(pico_sdk_import.cmake)

project(my_thesis_project C CXX)

# Initialize the Pico SDK
pico_sdk_init()

# Add your source files
add_executable(${PROJECT_NAME}
    src/main.cpp
    # you can add more .cpp/.c files here
)

# Link libraries you need
target_link_libraries(${PROJECT_NAME}
    pico_stdlib
    hardware_gpio
    hardware_spi
    hardware_dma
    # etc
)

# enable stdio over USB or UART
pico_enable_stdio_usb(${PROJECT_NAME} 1)
pico_enable_stdio_uart(${PROJECT_NAME} 0)

# produce UF2, map, etc
pico_add_extra_outputs(${PROJECT_NAME})
